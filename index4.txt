    <script>
        const defaultText = `花园里有一只漂亮的蝴蝶\n消灭所有怪物后才能过关\n我不瞒你说，这件事情.`;

        function segmentChineseText(text) {
            if (text.includes('|')) {
                return text.split('|').map(w => w.trim()).filter(w => w.length > 0);
            }
            if (typeof Intl !== 'undefined' && Intl.Segmenter) {
                const segmenter = new Intl.Segmenter('zh-CN', { granularity: 'word' });
                const segments = segmenter.segment(text);
                const results = [];
                for (const segment of segments) {
                    const word = segment.segment.trim();
                    if (word && word !== ' ') {
                        results.push(word);
                    }
                }
                return results;
            }
            return text.split('').filter(c => c.trim() !== '');
        }

        function getPinyin(text) {
            try {
                if (typeof pinyinPro !== 'undefined' && pinyinPro.pinyin) {
                    return pinyinPro.pinyin(text, { toneType: 'symbol' });
                }
            } catch (e) {}
            return '';
        }

        function parseSentences() {
            const raw = document.getElementById('rawSentences').value.trim();
            if (!raw) return alert("Vui lòng nhập danh sách câu!");

            const lines = raw.split('\n').map(l => l.trim()).filter(l => l.length > 0);
            const tbody = document.getElementById('parsedTbody');
            tbody.innerHTML = '';

            lines.forEach((line, qIdx) => {
                const words = segmentChineseText(line);
                words.forEach((word) => {
                    const py = getPinyin(word);
                    addRowToParsedTable(qIdx, word, py);
                });
            });

            buildExercises();
        }

        function addRowToParsedTable(qIdx, word, py) {
            const tbody = document.getElementById('parsedTbody');
            const tr = document.createElement('tr');
            tr.dataset.qidx = qIdx;
            tr.innerHTML = `
                <td><b>Câu ${qIdx + 1}</b></td>
                <td><input type="text" class="edit-hanzi" value="${word}"></td>
                <td><input type="text" class="edit-pinyin" value="${py}"></td>
                <td class="btn-del-row" onclick="this.closest('tr').remove()">✕</td>
            `;
            tbody.appendChild(tr);
        }

        function getParsedData() {
            const rows = document.querySelectorAll('#parsedTbody tr');
            const questionsDict = {};

            rows.forEach(tr => {
                const qIdx = tr.dataset.qidx;
                const hanzi = tr.querySelector('.edit-hanzi').value.trim();
                const pinyin = tr.querySelector('.edit-pinyin').value.trim();

                if (!questionsDict[qIdx]) questionsDict[qIdx] = [];
                if (hanzi) {
                    questionsDict[qIdx].push({ hanzi, pinyin });
                }
            });

            return Object.values(questionsDict);
        }

        function shuffleArray(array) {
            const arr = [...array];
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

        function buildExercises() {
            const questions = getParsedData();
            if (questions.length === 0) return;

            const showPinyin = document.getElementById('showPinyinOpt').value === 'true';
            const container = document.getElementById('exerciseContainer');
            container.innerHTML = '';

            questions.forEach((qWords, qIdx) => {
                const card = document.createElement('div');
                card.className = 'exercise-card';

                const shuffled = shuffleArray(qWords.map((item, id) => ({ ...item, originalId: id })));

                card.innerHTML = `
                    <div class="exercise-header">
                        <span class="q-number">Câu ${qIdx + 1}</span> Sắp xếp các từ thành câu hoàn chỉnh:
                    </div>
                    <div class="word-pool" id="pool-${qIdx}" ondragover="allowDrop(event)" ondrop="drop(event)"></div>
                    <div class="answer-zone" id="answer-${qIdx}" ondragover="allowDrop(event)" ondrop="drop(event)"></div>
                    <div class="correct-answer-box" id="correct-ans-${qIdx}">
                        <b>Đáp án đúng:</b> ${qWords.map(w => w.hanzi).join(' ')}
                    </div>
                `;

                container.appendChild(card);

                const pool = card.querySelector(`#pool-${qIdx}`);
                shuffled.forEach((item) => {
                    const chip = createWordChip(item, qIdx, showPinyin);
                    pool.appendChild(chip);
                });
            });

            document.getElementById('btnSubmit').style.display = 'block';
        }

        function createWordChip(item, qIdx, showPinyin) {
            const chip = document.createElement('div');
            chip.className = 'word-chip';
            chip.draggable = true;
            chip.dataset.hanzi = item.hanzi;
            chip.dataset.pinyin = item.pinyin;

            chip.innerHTML = `
                ${showPinyin && item.pinyin ? `<span class="chip-pinyin">${item.pinyin}</span>` : ''}
                <span class="chip-hanzi">${item.hanzi}</span>
            `;

            chip.ondragstart = (e) => { window.draggedElement = chip; };

            chip.onclick = () => {
                const pool = document.getElementById(`pool-${qIdx}`);
                const answer = document.getElementById(`answer-${qIdx}`);
                if (chip.parentElement === pool) {
                    answer.appendChild(chip);
                } else {
                    pool.appendChild(chip);
                }
            };

            return chip;
        }

        function allowDrop(e) { e.preventDefault(); }
        function drop(e) {
            e.preventDefault();
            if (window.draggedElement) {
                e.currentTarget.appendChild(window.draggedElement);
                window.draggedElement = null;
            }
        }

        function gradeExercise() {
            const questions = getParsedData();

            questions.forEach((qWords, qIdx) => {
                const answerZone = document.getElementById(`answer-${qIdx}`);
                const userChips = answerZone.querySelectorAll('.word-chip');
                const correctBox = document.getElementById(`correct-ans-${qIdx}`);

                userChips.forEach((chip, idx) => {
                    const expectedHanzi = qWords[idx] ? qWords[idx].hanzi : null;
                    const actualHanzi = chip.dataset.hanzi;

                    chip.classList.remove('correct', 'incorrect');
                    if (actualHanzi === expectedHanzi) {
                        chip.classList.add('correct');
                    } else {
                        chip.classList.add('incorrect');
                    }
                });

                correctBox.style.display = 'block';
            });
        }

        function generateShareLink() {
            const questions = getParsedData();
            const showPinyin = document.getElementById('showPinyinOpt').value;

            if (questions.length === 0) return alert("Chưa có dữ liệu bài tập!");

            const payload = { questions, showPinyin };

            if (typeof LZString !== 'undefined') {
                const compressed = LZString.compressToEncodedURIComponent(JSON.stringify(payload));
                const baseUrl = window.location.origin + window.location.pathname;
                const studentUrl = `${baseUrl}?z=${compressed}&s=1`;

                if (navigator.clipboard) {
                    navigator.clipboard.writeText(studentUrl).then(() => {
                        alert("✅ ĐÃ SAO CHÉP LINK BÀI TẬP BÀI HỌC SINH!\n\nLink này hoạt động vĩnh viễn, học sinh mở ra sẽ làm bài trực tiếp.");
                    });
                } else {
                    prompt("Sao chép link bài tập:", studentUrl);
                }
            }
        }

        window.onload = function() {
            const urlParams = new URLSearchParams(window.location.search);
            const compressed = urlParams.get('z');
            const isStudent = urlParams.get('s') === '1';

            if (isStudent) {
                document.body.classList.add('student-mode');
            }

            if (compressed && typeof LZString !== 'undefined') {
                try {
                    const decompressed = LZString.decompressFromEncodedURIComponent(compressed);
                    if (decompressed) {
                        const data = JSON.parse(decompressed);
                        document.getElementById('showPinyinOpt').value = data.showPinyin;

                        const tbody = document.getElementById('parsedTbody');
                        tbody.innerHTML = '';
                        data.questions.forEach((qWords, qIdx) => {
                            qWords.forEach(w => {
                                addRowToParsedTable(qIdx, w.hanzi, w.pinyin);
                            });
                        });
                        buildExercises();
                    }
                } catch (e) {
                    console.error("Lỗi giải nén dữ liệu:", e);
                }
            } else {
                document.getElementById('rawSentences').value = defaultText;
                parseSentences();
            }
        };
    </script>
